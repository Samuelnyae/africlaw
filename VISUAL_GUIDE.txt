  cd africlaw
  npm install
  npm run dev

  # Test health check
  curl http://localhost:3000/health

  # Test admin dashboard (requires auth)
  curl -u admin:admin http://localhost:3000/admin/data

  # View homepage
  open http://localhost:3000


📊 API ENDPOINTS (QUICK REFERENCE)
─────────────────────────────────────────────────────────────────────────────

  GET  /                          → Homepage (API docs)
  GET  /health                    → Server status
  POST /whatsapp/webhook          → Receive messages
  GET  /admin                     → Admin dashboard
  GET  /admin/data                → Dashboard metrics
  GET  /api/users                 → List users
  GET  /api/conversations/:phone  → User messages
  POST /mpesa/callback            → Payment confirmation
